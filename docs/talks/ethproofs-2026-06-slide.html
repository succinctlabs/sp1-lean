EthProofs #9

SP1 in Clean: modular and expressive

sp1-lean proved SP1's RISC-V chips sound to the Sail spec — per chip. We rebuilt it on Clean as a modular chain of small, composable, separately-proved artifacts — and now prove completeness too, sorry-free, for all 25 chips.

circuit · main — composes subcircuits (MulChip)
def main (input : Var Inputs (ZMod p)) :
    Circuit (ZMod p) (Var MulCols (ZMod p)) := do
  let b := input.op_b_val; let c := input.op_c_val
  -- CPU-state reader subcircuit
  assertion CPUState.circuit ⟨…⟩
  -- witness the 5 variant flags
  let flags ← witnessVector 5 (hintFlags …)
  -- the Mul gadget: witnessed cols + subcircuit
  let cols ← witness (MulOperation.populate b c …)
  assertion MulOperation.circuit ⟨b, c, cols, …⟩
  -- flag-weighted result word
  let value ← witnessVector 4 (aSelector cols flags)
  -- register-reader subcircuit
  assertion RTypeReader.circuit ⟨…, value⟩
  -- is_real ∈ {0,1}
  is_real * (is_real - 1) === 0
  return ⟨…, cols, value, flags⟩
contract · Assumptions + Constraints ⟹ Spec
def Assumptions (input : Inputs (ZMod p)) : Prop :=
  Word.isU64 input.op_b_val ∧ Word.isU64 input.op_c_val

def Spec (input : Inputs (ZMod p))
    (cols : MulCols (ZMod p)) : Prop :=
  let value := Word.toBitVec64 cols.a
  let b := Word.toBitVec64 input.op_b_val
  let c := Word.toBitVec64 input.op_c_val
  -- the reader's own spec
  RTypeReader.Spec {…} ∧
  -- is_real is binary
  (input.is_real = 0 ∨ input.is_real = 1) ∧
  -- the RV64 meaning, per committed flag
  (input.is_real = 1 →
    (cols.is_mul    = 1 → value = RV64.mul    b c) ∧
    (cols.is_mulh   = 1 → value = RV64.mulh   b c) ∧
    (cols.is_mulhu  = 1 → value = RV64.mulhu  b c) ∧
    (cols.is_mulhsu = 1 → value = RV64.mulhsu b c) ∧
    (cols.is_mulw   = 1 → value = RV64.mulw   b c))
